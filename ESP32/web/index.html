Deploy Front-End to ESP32 Flash (SPIFFS / LittleFS)

Copy this Arduino C++ code snippet to serve this entire dashboard directly from your ESP32 asynchronously.

#include <WiFi.h>
#include <ESPAsyncWebServer.h>
#include <AsyncTCP.h>
#include <ArduinoJson.h>

AsyncWebServer server(80);
AsyncWebSocket ws("/ws");

void onEvent(AsyncWebSocket *server, AsyncWebSocketClient *client, AwsEventType type, void *arg, uint8_t *data, size_t len) {
  if (type == WS_EVT_DATA) {
    StaticJsonDocument<200> doc;
    deserializeJson(doc, data, len);
    
    if (doc.containsKey("pin")) {
      int pin = doc["pin"];
      int state = doc["state"];
      pinMode(pin, OUTPUT);
      digitalWrite(pin, state);
    }
  }
}

void setup() {
  Serial.begin(115200);
  WiFi.begin("YOUR_SSID", "YOUR_PASSWORD");
  
  ws.onEvent(onEvent);
  server.addHandler(&ws);
  
  // Serve single index.html from LittleFS or Inline String
  server.on("/", HTTP_GET, [](AsyncWebServerRequest *request){
    request->send(200, "text/html", "LOAD_HTML_STRING_HERE");
  });

  server.begin();
}

void loop() {
  ws.cleanupClients();
  // Send telemetry periodic JSON to ws.textAll(...)
}
